"""Original synthesized transition SFX (no licensing): whoosh, low boom hit, riser, sub drop, tick. 48 kHz stereo wav in sfx/."""
import numpy as np, soundfile as sf, os
SR = 48000; D = "/home/claude/doc/chiricahua/sfx"; os.makedirs(D, exist_ok=True)
rng = np.random.default_rng(11)
def bandnoise(n, lo, hi):
    x = rng.standard_normal(n); X = np.fft.rfft(x); f = np.fft.rfftfreq(n, 1 / SR)
    X[(f < lo) | (f > hi)] = 0; return np.fft.irfft(X, n).astype(np.float32)
def norm(x, peak=0.9): return (x / (np.max(np.abs(x)) + 1e-9) * peak).astype(np.float32)
def st(x, w=0.0):  # stereo with slight width
    d = int(0.012 * SR * w); r = np.roll(x, d) if d else x; return np.stack([x, r], 1)
def save(name, x): sf.write(f"{D}/{name}.wav", x, SR)

# whoosh: filtered noise with moving band + swell envelope, pan sweep
n = int(1.2 * SR); t = np.linspace(0, 1, n)
w = np.zeros(n, np.float32)
for k in range(6):
    seg = bandnoise(n, 300 + k * 500, 900 + k * 900) * np.exp(-((t - (0.35 + k * 0.05)) ** 2) / 0.02)
    w += seg
env = np.sin(np.pi * np.clip(t / 0.85, 0, 1)) ** 2; w = norm(w * env, 0.7)
pan = np.clip(t * 1.2, 0, 1); save("whoosh", np.stack([w * (1 - 0.5 * pan), w * (0.5 + 0.5 * pan)], 1))
# soft whoosh (shorter, darker)
n2 = int(0.8 * SR); t2 = np.linspace(0, 1, n2)
w2 = bandnoise(n2, 200, 2200) * np.sin(np.pi * t2) ** 3; save("whoosh_soft", st(norm(w2, 0.5), 1))
# low boom hit: sine sweep 70->38 Hz + noise thump, long tail
n3 = int(3.0 * SR); t3 = np.arange(n3) / SR
f = 38 + 32 * np.exp(-t3 * 6); ph = 2 * np.pi * np.cumsum(f) / SR
boom = np.sin(ph) * np.exp(-t3 * 1.6) + 0.35 * bandnoise(n3, 40, 300) * np.exp(-t3 * 9)
save("hit_low", st(norm(boom, 0.85), 0.5))
# riser: rising filtered noise + rising tone, 3 s
n4 = int(3.0 * SR); t4 = np.linspace(0, 1, n4)
tone = np.sin(2 * np.pi * np.cumsum(110 + 330 * t4 ** 2) / SR) * 0.3
rise = (bandnoise(n4, 800, 6000) * 0.6 + tone) * t4 ** 2.2
rise[-int(0.05 * SR):] *= np.linspace(1, 0, int(0.05 * SR)); save("riser", st(norm(rise, 0.6), 1))
# sub drop (for reveals)
n5 = int(2.2 * SR); t5 = np.arange(n5) / SR
drop = np.sin(2 * np.pi * np.cumsum(90 * np.exp(-t5 * 1.5) + 30) / SR) * np.exp(-t5 * 1.2)
save("sub_drop", st(norm(drop, 0.8)))
# clock tick / date stamp tick
n6 = int(0.08 * SR); t6 = np.arange(n6) / SR
tick = bandnoise(n6, 2000, 7000) * np.exp(-t6 * 120); save("tick", st(norm(tick, 0.5)))
# typewriter-like keys (for date stamps): 6 ticks
n7 = int(0.9 * SR); tw = np.zeros(n7, np.float32)
for i, ofs in enumerate([0.0, 0.11, 0.2, 0.33, 0.45, 0.56]):
    a = int(ofs * SR); tw[a:a + n6] += tick * (0.8 + 0.2 * rng.random())
save("type", st(norm(tw, 0.45)))
print(sorted(os.listdir(D)))

# ---- ambiences (60 s loops) ----
L = 60 * SR; tt = np.arange(L) / SR
def lfo(rate, depth, ph=0): return 1 + depth * np.sin(2 * np.pi * rate * tt + ph)
# wind: low/mid band noise with slow gusts, stereo decorrelated
wl = bandnoise(L, 80, 900) * lfo(0.07, 0.5) * lfo(0.19, 0.25, 1.3)
wr = bandnoise(L, 80, 900) * lfo(0.07, 0.5, 0.6) * lfo(0.17, 0.25, 2.1)
gust = bandnoise(L, 500, 2500) * np.clip(np.sin(2 * np.pi * 0.05 * tt) , 0, 1) ** 3 * 0.4
save("amb_wind", np.stack([norm(wl + gust, 0.5), norm(wr + gust, 0.5)], 1))
# fire crackle: rumble + random pops
rum = bandnoise(L, 40, 400) * 0.6
pops = np.zeros(L, np.float32)
for p in rng.integers(0, L - 2000, 1400):
    k = int(rng.integers(200, 1600)); pops[p:p + k] += bandnoise(k, 1500, 9000) * np.exp(-np.arange(k) / (k / 5)) * rng.uniform(0.2, 1.0)
save("amb_fire", st(norm(rum * lfo(0.3, 0.3) + pops * 0.8, 0.6), 1))
# gallop: 3-beat hoof pattern on dirt
gal = np.zeros(L, np.float32); beat = 0.55; t0 = 0.0
thump = bandnoise(int(0.09 * SR), 60, 900) * np.exp(-np.arange(int(0.09 * SR)) / (0.018 * SR))
while t0 < 59:
    for off in (0.0, 0.11, 0.21):
        a = max(0, int((t0 + off + rng.normal(0, 0.006)) * SR)); gal[a:a + len(thump)] += thump * rng.uniform(0.7, 1.0)
    t0 += beat
save("amb_gallop", st(norm(gal + bandnoise(L, 100, 600) * 0.05, 0.6), 1))
print("ambiences ok")

    p = subprocess.run(["ffmpeg", "-v", "error", "-i", f"{D}/narr/{id_}.m4a", "-f", "f32le", "-ac", "1", "-ar", str(SR), "-"], capture_output=True)
    return np.frombuffer(p.stdout, np.float32)

def load_clip_audio(src, ss, dur):
    p = subprocess.run(["ffmpeg", "-v", "error", "-ss", str(ss), "-t", str(dur), "-i", src, "-vn", "-af",
                        "highpass=f=80,loudnorm=I=-26:TP=-6:LRA=7,afade=t=in:d=0.25,afade=t=out:st=%.2f:d=0.4" % max(0, dur - 0.4),
                        "-f", "f32le", "-ac", "1", "-ar", str(SR), "-"], capture_output=True)
    a = np.frombuffer(p.stdout, np.float32)
    n = int(dur * SR)
    return np.pad(a, (0, max(0, n - len(a))))[:n]

audio = []; t = 0.0; tl = []; mask = []
# KEEP_TIMING: keep every line's slot from a previous timeline; a re-voiced line that grew gets whole extra frames only,
# so all other shots keep identical durations (cached renders stay valid)
KEEP = {it["id"]: it["dur"] for it in json.load(open(os.environ["KEEP_TIMING"])) if it.get("id")} if os.environ.get("KEEP_TIMING") else {}
import math
def push(x, voice=True):
    global t
    audio.append(x); mask.append(np.full(len(x), voice, bool)); t += len(x) / SR
def silence(s): push(np.zeros(int(s * SR), np.float32), False)
silence(0.6)
for s in segs:
    if s["id"].startswith("h"):
        if s["chapter"] == "cold_open": continue
        tl.append({"kind": "card", "id": s["id"], "chapter": s["chapter"], "heading": s["heading"], "start": round(t, 3), "dur": CARD})
        silence(CARD); continue
    a = load_narr(s["id"]); st = t; push(a); g = GAP + BREATH.get(s["id"], 0)
    if s["id"] in KEEP:
        nat = len(a) / SR + g; old = KEEP[s["id"]]
        g += (old - nat) if nat <= old + 0.01 else (math.ceil((nat - old) * 30 - 1e-6) / 30 + old - nat)
    silence(g)
    tl.append({"kind": "seg", "id": s["id"], "chapter": s["chapter"], "start": round(st, 3), "dur": round(len(a) / SR + g, 3), "text": s["text"], "cues": s["cues"]})
    if s["id"] == "s008":
        tl.append({"kind": "title", "start": round(t, 3), "dur": COLD_TITLE}); silence(COLD_TITLE)
    if s["id"] in INS:
        ins = INS[s["id"]]
        st = t; silence(PAD)
        push(load_clip_audio(ins["src"], ins["ss"], ins["dur"]))
        silence(PAD)

        tl.append({"kind": "insert", "id": s["id"] + "_ins", "chapter": s["chapter"], "start": round(st, 3), "dur": round(ins["dur"] + 2 * PAD, 3),
                   "src": ins["src"], "ss": max(0, ins["ss"] - PAD), "label": ins.get("label", ""), "sub": ins.get("sub", "")})
