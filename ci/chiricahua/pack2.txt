def load(p):

def place(buf, a, t0, gain, fade=0.05):
    s = int(t0 * SR);
    if s >= N: return
    a = a[: N - max(s, 0)].copy() if s >= 0 else a[-s:N - 0].copy()
    s = max(s, 0); n = len(a)
    if n == 0: return

    if k: r = np.linspace(0, 1, k, dtype=np.float32); a[:k] *= r[:, None]; a[-k:] *= r[::-1, None]
    buf[s:s + n] += a * gain

#@@FILE chiricahua/score.py
"""Procedural ambient documentary score (original, royalty-free). Writes WAV beds per mood.
moods: wonder (D major/lydian pads), tension (low drone, minor clusters, pulse), somber (minor pads, sparse piano), western (open fifths, slow guitar-like plucks)."""
import numpy as np, sys, subprocess, os
SR = 48000
rng = np.random.default_rng(11)

def note(m): return 440.0 * 2 ** ((m - 69) / 12)

def lowpass(x, cutoff):
    # one-pole lowpass, cutoff in Hz (scalar)
    a = np.exp(-2 * np.pi * cutoff / SR); y = np.empty_like(x); acc = 0.0
    b = 1 - a
    for i in range(0, len(x), 4096):
        seg = x[i:i + 4096]
        from scipy.signal import lfilter
        y_seg, zf = lfilter([b], [1, -a], seg, zi=[acc * a]); y[i:i + 4096] = y_seg; acc = y_seg[-1]
    return y

def pad(freqs, dur, bright=900, detune=0.003, vol=0.18):
    t = np.arange(int(dur * SR)) / SR; out = np.zeros_like(t)
    for f in freqs:
        for d in (-detune, 0, detune):
            ph = rng.uniform(0, 2 * np.pi)
            saw = 2 * ((t * f * (1 + d) + ph / (2 * np.pi)) % 1) - 1
            out += saw * 0.33
    from scipy.signal import butter, sosfilt
    sos = butter(2, bright, "low", fs=SR, output="sos"); out = sosfilt(sos, out)
    env = np.minimum(1, t / 3.0) * np.minimum(1, (dur - t) / 3.0)
    return out / (len(freqs) * 1.2) * env * vol

def pluck(f, dur=4.0, vol=0.22, decay=2.2):
    t = np.arange(int(dur * SR)) / SR
    x = (np.sin(2 * np.pi * f * t) + 0.35 * np.sin(2 * np.pi * 2 * f * t) + 0.12 * np.sin(2 * np.pi * 3 * f * t))
    return x * np.exp(-t * decay) * vol * np.minimum(1, t / 0.005)

def reverb(x, wet=0.35):
    from scipy.signal import fftconvolve
    n = int(SR * 3.2); t = np.arange(n) / SR
    ir = rng.normal(0, 1, n) * np.exp(-t * 2.3); ir[0] = 0
    y = fftconvolve(x, ir)[:len(x)]; y /= (np.max(np.abs(y)) + 1e-9)
    return x * (1 - wet) + y * wet * np.max(np.abs(x))

def wind(dur, vol=0.03):
    from scipy.signal import butter, sosfilt
    n = rng.normal(0, 1, int(dur * SR)); sos = butter(2, [200, 900], "band", fs=SR, output="sos")
    mod = 0.6 + 0.4 * np.sin(2 * np.pi * np.arange(len(n)) / SR / 9.0)
    return sosfilt(sos, n) * mod * vol

def place(buf, x, at):
    i = int(at * SR); j = min(len(buf), i + len(x)); buf[i:j] += x[:j - i]

def build(mood, dur=240):
    N = int(dur * SR); L = np.zeros(N); R = np.zeros(N)
    if mood == "wonder":
        prog = [[50, 57, 62, 66, 69], [47, 54, 59, 62, 66], [43, 50, 55, 59, 62], [45, 52, 57, 61, 64]]  # D, Bm, G, A
        seg = 12.0; bright = 1400; pluck_scale = [62, 64, 66, 69, 71, 74]; pl_every = 1.5
    elif mood == "tension":
        prog = [[38, 45, 50, 53], [38, 44, 50, 53], [36, 43, 48, 51], [37, 44, 49, 52]]
        seg = 10.0; bright = 600; pluck_scale = [50, 53, 56]; pl_every = 3.0
    elif mood == "somber":
        prog = [[45, 52, 57, 60, 64], [41, 48, 53, 57, 60], [43, 50, 55, 59, 62], [40, 47, 52, 55, 59]]  # Am F G Em
        seg = 12.0; bright = 900; pluck_scale = [57, 60, 64, 67, 69]; pl_every = 2.5
    else:  # western
        prog = [[40, 47, 52, 59], [45, 52, 57, 64], [43, 50, 55, 62], [40, 47, 52, 59]]
        seg = 11.0; bright = 1100; pluck_scale = [52, 55, 57, 59, 62, 64]; pl_every = 1.8
    t = 0.0; k = 0
    while t < dur:
        ch = prog[k % len(prog)]
        p = pad([note(m) for m in ch], seg + 4, bright=bright)
        place(L, p, t); place(R, p * 0.95, t + 0.02)
        k += 1; t += seg
    # drone
    tt = np.arange(N) / SR; root = note(prog[0][0] - 12)
    drone = (np.sin(2 * np.pi * root * tt) * 0.06 + np.sin(2 * np.pi * root * 1.5 * tt) * 0.02) * (0.7 + 0.3 * np.sin(2 * np.pi * tt / 20))
    L += drone; R += drone
    if mood == "tension":
        beat = np.zeros(N)
        for s in np.arange(4, dur - 2, 1.6):
            th = pluck(note(26), 1.2, vol=0.35, decay=6); place(beat, th, s)
        L += beat; R += beat
    s = 3.0
    while s < dur - 5:
        if rng.random() < 0.75:
            f = note(int(rng.choice(pluck_scale)))
            pl = pluck(f, 4.0, vol=0.10 if mood != "western" else 0.13, decay=1.6 if mood != "tension" else 2.5)
            pan = rng.uniform(0.3, 0.7); place(L, pl * (1 - pan), s); place(R, pl * pan, s)
        s += pl_every * rng.uniform(0.8, 1.6)
    w = wind(dur, 0.02 if mood != "tension" else 0.03); L += w; R += w[::-1]
    L = reverb(L, 0.4); R = reverb(R, 0.4)
    fade = np.minimum(1, np.minimum(tt / 4, (dur - tt) / 4))
    st = np.stack([L * fade, R * fade], 1); st /= np.max(np.abs(st)) * 1.12
    return st.astype(np.float32)

if __name__ == "__main__":
    import soundfile as sf
    out = sys.argv[1]; os.makedirs(out, exist_ok=True)
    for m in (sys.argv[2:] or ["wonder", "tension", "somber", "western"]):
        sf.write(f"{out}/{m}.wav", build(m, 240), SR); print("score", m, flush=True)
