#@@FILE chiricahua/engine.py
"""Documentary render engine: shot plan -> 1920x1080 episode.
Shot types:
  video: {"type":"video","src":path,"ss":s,"dur":d,"speed":1.0,"drift":0.04,"fx":0.5,"fy":0.5}
  photo: {"type":"photo","src":path,"dur":d,"z0":1.0,"z1":1.12,"x0":.5,"y0":.5,"x1":.5,"y1":.45}
  card:  {"type":"card","dur":d,"title":"...","sub":"...","style":"chapter|date|title"}
  map:   {"type":"map","src":mp4} (pre-rendered)
Common: "label": lower-third text, "date": corner date stamp.
"""
import os, subprocess, json, math, hashlib
from PIL import Image, ImageDraw, ImageFont, ImageFilter
import numpy as np

W, H, FPS = 1920, 1080, 30
FD = "/usr/share/fonts/opentype/inter/"
SERIF = None
for p in ["/usr/share/fonts/truetype/dejavu/DejaVuSerif.ttf", "/usr/share/fonts/truetype/liberation/LiberationSerif-Regular.ttf"]:
    if os.path.exists(p): SERIF = p; break
def F(w, s):
    return ImageFont.truetype(FD + {"black": "InterDisplay-Black.otf", "bold": "Inter-Bold.otf", "semi": "Inter-SemiBold.otf",
                                    "med": "Inter-Medium.otf", "light": "Inter-Light.otf" if os.path.exists(FD + "Inter-Light.otf") else "Inter-Medium.otf"}[w], s)
def SF(s, bold=False):
    for p in (["/usr/share/fonts/truetype/dejavu/DejaVuSerif-Bold.ttf"] if bold else []) + [SERIF]:
        if p and os.path.exists(p): return ImageFont.truetype(p, s)
    return F("bold", s)

GRADE = "eq=contrast=1.06:saturation=1.05:gamma=0.98,vignette=PI/5"
CACHE = os.environ.get("ENGINE_CACHE", "/home/claude/doc/chiricahua/cache"); os.makedirs(CACHE, exist_ok=True)

def run(cmd):
    p = subprocess.run(cmd, capture_output=True, text=True)
    if p.returncode: raise RuntimeError(p.stderr[-1500:])

def key(shot):
    return hashlib.md5(json.dumps(shot, sort_keys=True).encode()).hexdigest()[:12]

def overlay_png(shot):
    """Lower-third label and/or date stamp as transparent PNG (None if nothing)."""
    if not shot.get("label") and not shot.get("date"): return None
    im = Image.new("RGBA", (W, H), (0, 0, 0, 0)); d = ImageDraw.Draw(im)
    if shot.get("date"):
        f = F("semi", 42); t = shot["date"].upper()
        tw = d.textlength(t, font=f)
        d.rectangle([90, 84, 90 + 7, 84 + 56], fill=(214, 160, 82, 255))
        d.text((112, 88), t, font=f, fill=(245, 238, 225, 255), stroke_width=2, stroke_fill=(0, 0, 0, 140))
    if shot.get("label"):
        f = F("med", 34); t = shot["label"]
        tw = d.textlength(t, font=f)
        x, y = 90, H - 130
        g = Image.new("RGBA", (int(tw + 60), 64), (0, 0, 0, 0)); gd = ImageDraw.Draw(g)
        gd.rounded_rectangle([0, 0, g.width, g.height], 8, fill=(10, 10, 12, 150))
        im.alpha_composite(g, (x - 20, y - 10))
        d.text((x + 10, y), t, font=f, fill=(235, 230, 220, 255))
    p = f"{CACHE}/ov_{key({'l': shot.get('label'), 'd': shot.get('date')})}.png"; im.save(p); return p

def card_png(shot):
    im = Image.new("RGB", (W, H), (12, 11, 10)); d = ImageDraw.Draw(im)
    # subtle warm radial
    arr = np.zeros((H, W, 3), np.float32)
    yy, xx = np.mgrid[0:H, 0:W]; r = np.sqrt(((xx - W / 2) / W) ** 2 + ((yy - H / 2) / H) ** 2)
    glow = np.clip(1 - r * 1.6, 0, 1)[..., None] * np.array([46, 34, 22], np.float32)
    im = Image.fromarray(np.clip(arr + 12 + glow, 0, 255).astype(np.uint8)); d = ImageDraw.Draw(im)
    st = shot.get("style", "chapter")
    if st == "chapter":
        if shot.get("kicker"): d.text((W / 2, H / 2 - 120), shot["kicker"].upper(), font=F("semi", 30), fill=(214, 160, 82), anchor="mm")
        fs = 84
        while d.textlength(shot["title"], font=SF(fs, True)) > W - 240 and fs > 40: fs -= 4
        d.text((W / 2, H / 2 - 20), shot["title"], font=SF(fs, True), fill=(240, 232, 218), anchor="mm")
        d.line([(W / 2 - 60, H / 2 + 60), (W / 2 + 60, H / 2 + 60)], fill=(214, 160, 82), width=3)
        if shot.get("sub"): d.text((W / 2, H / 2 + 110), shot["sub"], font=F("med", 34), fill=(190, 182, 168), anchor="mm")
    elif st == "date":
        d.text((W / 2, H / 2), shot["title"], font=SF(96, True), fill=(240, 232, 218), anchor="mm")
        if shot.get("sub"): d.text((W / 2, H / 2 + 90), shot["sub"].upper(), font=F("semi", 30), fill=(214, 160, 82), anchor="mm")
    elif st == "title":
        d.text((W / 2, H / 2 - 30), shot["title"], font=SF(120, True), fill=(244, 236, 220), anchor="mm")
        if shot.get("sub"): d.text((W / 2, H / 2 + 80), shot["sub"], font=F("semi", 38), fill=(214, 160, 82), anchor="mm")
    p = f"{CACHE}/card_{key(shot)}.png"; im.save(p); return p

def render_shot(shot, out):
    dur = shot["dur"]; t = shot["type"]; frames = int(round(dur * FPS))
    ov = overlay_png(shot)
    fades = []
    if shot.get("fin"): fades.append(f"fade=t=in:st=0:d={shot['fin']}")
    if shot.get("fout"): fades.append(f"fade=t=out:st={max(0, dur - shot['fout']):.3f}:d={shot['fout']}")
    fade = ",".join(fades) if fades else None
    vf_tail = [GRADE, f"fps={FPS}", "format=yuv420p"]
    if t == "video":
        sp = shot.get("speed", 1.0); drift = shot.get("drift", 0.035)
        # cover-crop to 16:9 around focus, then slow push-in drift
        fx, fy = shot.get("fx", 0.5), shot.get("fy", 0.5)
        z = f"(1+{drift}*on/{max(frames,1)})"
        vf = [f"setpts=PTS/{sp}", "tpad=stop_mode=clone:stop_duration=60"] + ([shot["vf_extra"]] if shot.get("vf_extra") else []) + [ f"scale={W*2}:{H*2}:force_original_aspect_ratio=increase",
              f"crop={W*2}:{H*2}:(in_w-{W*2})*{fx}:(in_h-{H*2})*{fy}",
              f"zoompan=z='{z}':x='iw/2-(iw/zoom/2)':y='ih/2-(ih/zoom/2)':d=1:s={W}x{H}:fps={FPS}"] + vf_tail
        if fade: vf.append(fade)
        inp = ["-ss", str(shot.get("ss", 0)), "-t", str(dur * sp + 0.5), "-i", shot["src"]]
    elif t == "photo":
        z0, z1 = shot.get("z0", 1.0), shot.get("z1", 1.12)
        x0, y0, x1, y1 = shot.get("x0", .5), shot.get("y0", .5), shot.get("x1", .5), shot.get("y1", .5)
        e = f"(on/{frames})"; ez = f"(0.5-0.5*cos(PI*{e}))"
        z = f"({z0}+({z1}-{z0})*{ez})"
        x = f"(({x0}+({x1}-{x0})*{ez})*iw-iw/zoom/2)"; y = f"(({y0}+({y1}-{y0})*{ez})*ih-ih/zoom/2)"
        bg = shot.get("bg", "cover")
        if bg == "cover":
            pre = f"scale={W*3}:{H*3}:force_original_aspect_ratio=increase,crop={W*3}:{H*3}"
        else:  # fit with blurred fill (for portraits / documents)
            pre = (f"split[a][b];[a]scale={W*3}:{H*3}:force_original_aspect_ratio=increase,crop={W*3}:{H*3},boxblur=40:2,eq=brightness=-0.15[bgx];"
                   f"[b]scale={W*3}:{H*3}:force_original_aspect_ratio=decrease[fg];[bgx][fg]overlay=(W-w)/2:(H-h)/2")
        vf = [pre, f"zoompan=z='{z}':x='max(0,min(iw-iw/zoom,{x}))':y='max(0,min(ih-ih/zoom,{y}))':d={frames}:s={W}x{H}:fps={FPS}"] + vf_tail
        if shot.get("sepia"): vf.insert(-2, "colorchannelmixer=.393:.769:.189:0:.349:.686:.168:0:.272:.534:.131")
        if fade: vf.append(fade)
        inp = ["-loop", "1", "-t", str(dur + 0.2), "-i", shot["src"]]
    elif t == "card":
        png = card_png(shot)
        vf = [f"zoompan=z='1+0.03*on/{frames}':d={frames}:s={W}x{H}:fps={FPS}", "format=yuv420p",
              f"fade=t=in:st=0:d=0.6", f"fade=t=out:st={dur-0.6}:d=0.6"]
        inp = ["-loop", "1", "-t", str(dur + 0.2), "-i", png]
    elif t == "map":
        vf = ["tpad=stop_mode=clone:stop_duration=60", f"scale={W}:{H}", f"fps={FPS}", "format=yuv420p"] + ([fade] if fade else [])
        inp = ["-i", shot["src"]]
    else:
        raise ValueError(t)
    chain = ",".join(vf)
    if ov:
        fc = f"[0:v]{chain}[v0];[v0][1:v]overlay=0:0:format=auto,format=yuv420p[v]"
        cmd = ["ffmpeg", "-v", "error", "-y"] + inp + ["-loop", "1", "-i", ov, "-filter_complex", fc, "-map", "[v]"]
    else:
        cmd = ["ffmpeg", "-v", "error", "-y"] + inp + ["-vf", chain]
    cmd += ["-frames:v", str(frames), "-an", "-c:v", "libx264", "-preset", "veryfast", "-crf", "18", "-r", str(FPS), out]
    run(cmd)
    return out

def render_all(shots, workdir):
    os.makedirs(workdir, exist_ok=True); files = []
    for i, s in enumerate(shots):
        f = f"{workdir}/{i:04d}_{key(s)}.mp4"
        if not os.path.exists(f):
            render_shot(s, f)
        files.append(f)
    lst = f"{workdir}/concat.txt"; open(lst, "w").write("".join(f"file '{f}'\n" for f in files))
    return lst

def mix_audio(narr_wav, music, out, music_db=-20, duck=True):
    """music bed ducked under narration."""
    if music:
        fc = (f"[1:a]volume={10**(music_db/20):.3f},aloop=loop=-1:size=2e9[m];"
              f"[m][0:a]sidechaincompress=threshold=0.03:ratio=6:attack=20:release=400[md];"
              f"[0:a][md]amix=inputs=2:duration=first:normalize=0,loudnorm=I=-16:TP=-1.5[a]")
        run(["ffmpeg", "-v", "error", "-y", "-i", narr_wav, "-i", music, "-filter_complex", fc, "-map", "[a]", "-ar", "48000", out])
    else:
        run(["ffmpeg", "-v", "error", "-y", "-i", narr_wav, "-af", "loudnorm=I=-16:TP=-1.5", "-ar", "48000", out])

def final(concat_list, audio, out):
    run(["ffmpeg", "-v", "error", "-y", "-f", "concat", "-safe", "0", "-i", concat_list, "-i", audio,
         "-map", "0:v", "-map", "1:a", "-c:v", "libx264", "-preset", "medium", "-crf", "19", "-pix_fmt", "yuv420p",
         "-c:a", "aac", "-b:a", "192k", "-shortest", "-movflags", "+faststart", "-map_metadata", "-1", out])

#@@FILE chiricahua/make_dub_es.py
"""Spanish dub track (rule: main audio English, optional Spanish track).
Places each Spanish narration line in the exact slot of its English line (same timeline, same picture),
time-fitting with atempo when Spanish runs long; interview/news bites keep their original voices; cards stay silent.
Writes narration_full_es.wav + voice_mask_es.npy, then render_v2.py (MIX_TAG=_es, audio) builds the same music/ambience/SFX bed."""
import json, os, subprocess, sys, numpy as np, soundfile as sf
D = "/home/claude/doc/chiricahua"; SR = 48000
VOICE = sys.argv[1] if len(sys.argv) > 1 else "em_alex"; SRC = f"/home/claude/fb/chiricahua-narration-es/{VOICE}"
tl = json.load(open(f"{D}/timeline.json")); en, _ = sf.read(f"{D}/narration_full.wav", dtype="float32")
if en.ndim > 1: en = en.mean(1)
N = len(en); out = np.zeros(N, np.float32); mask = np.zeros(N, bool); report = []
def dec(p, af=None):
    cmd = ["ffmpeg", "-v", "error", "-i", p] + (["-af", af] if af else []) + ["-f", "f32le", "-ac", "1", "-ar", str(SR), "-"]
    return np.frombuffer(subprocess.run(cmd, capture_output=True).stdout, np.float32)
def trim(a, thr=10 ** (-45 / 20)):
    idx = np.where(np.abs(a) > thr)[0]
    return a[max(0, idx[0] - 1200): idx[-1] + 2400] if len(idx) else a
items = sorted(tl, key=lambda it: it["start"])
for k, it in enumerate(items):
    s0 = int(it["start"] * SR); s1 = int((it["start"] + it["dur"]) * SR)
    if it["kind"] == "insert":
        out[s0:s1] = en[s0:s1]; mask[s0:s1] = True; continue
    if it["kind"] != "seg": continue
    sp = f"{SRC}/{it['id']}.m4a"; fx = sp.replace("chiricahua-narration-es/", "chiricahua-narration-es-fix5/")
    if os.path.exists(fx): sp = fx                       # corrected line (fact-check) wins
    a = trim(dec(sp))
    # the slot: this line's time, plus the silent tail of the next item if it is a card/title (never into another voice)
    slot = (s1 - s0) - int(0.12 * SR)
    nxt = items[k + 1] if k + 1 < len(items) else None
    if nxt and nxt["kind"] in ("card", "title"): slot += int(min(1.2, nxt["dur"] * 0.3) * SR)
    r = len(a) / slot
    if r > 1.0:
        a = trim(dec(sp, f"atempo={min(r, 1.35):.4f}"))
        if len(a) > slot: report.append((it["id"], round(r, 2), "TRUNC", round((len(a) - slot) / SR, 2))); a = a[:slot].copy(); a[-2400:] *= np.linspace(1, 0, 2400)
    report.append((it["id"], round(r, 3)))
    n = min(len(a), N - s0); out[s0:s0 + n] = a[:n]; mask[s0:s0 + n] = np.abs(a[:n]) > 1e-4
# match narrator loudness to the English narration
def rms(x): return float(np.sqrt(np.mean(x[np.abs(x) > 1e-3] ** 2)))
g = rms(en) / max(rms(out), 1e-6); out *= g
sf.write(f"{D}/narration_full_es.wav", out, SR, subtype="FLOAT")
vm = mask[: N // (SR // 100) * (SR // 100)].reshape(-1, SR // 100).any(1)
np.save(f"{D}/voice_mask_es.npy", vm)
fast = [x for x in report if len(x) == 2 and x[1] > 1.12]; trunc = [x for x in report if len(x) == 4]
print("lines:", sum(1 for x in report if len(x) == 2), "| sped up >12%:", len(fast), fast[:12], "| truncated:", trunc, "| gain", round(g, 2))

def load(p):

def caption_cues(text, a, t0):
    """phrases of <= 4 words, sentence starts anchored to real pauses"""
    hop = SR // 100; env = np.sqrt(np.mean(a[: len(a) // hop * hop].reshape(-1, hop) ** 2, 1)); thr = 10 ** (-40 / 20)
    sil = env < thr; P_ = []; n = 0
    for k, s in enumerate(sil):
        if s: n += 1
        else:
            if n >= 14: P_.append((k - n / 2) / 100)
            n = 0
    sents = [x for x in re.split(r"(?<=[.?!])\s+", text.strip()) if x]
    L = np.cumsum([len(x) for x in sents]); tot = L[-1]; s0 = 0.0; s1 = len(a) / SR; cuts = []
    for k in range(len(sents) - 1):
        g = s1 * L[k] / tot; c = [p for p in P_ if (not cuts or p > cuts[-1] + 0.3)]
        b = min(c, key=lambda p: abs(p - g)) if c else g; cuts.append(b if abs(b - g) < 1.8 else g)
    T = [s0] + cuts + [s1]; cues = []
    for k, s in enumerate(sents):
        w = s.split(); ph_ = []; cur = []
        for x in w:
            cur.append(x)
            if len(cur) >= 4 or (len(cur) >= 2 and x[-1] in ",.?!:"): ph_.append(" ".join(cur)); cur = []
        if cur: ph_.append(" ".join(cur))
        a_, b_ = T[k] + 0.05, T[k + 1] - 0.05; tt = sum(len(p) for p in ph_); t = a_
        for p in ph_:
            d = (b_ - a_) * len(p) / tt; cues.append((t0 + t, t0 + t + d, p)); t += d
    return cues

def pauses(a, b):
    """midpoints (s) of silent runs >= 0.16 s strictly inside [a, b]"""
    i0, i1 = int(a * 100), int(b * 100); q = env[i0:i1] < thr; out = []; n = 0
    for k, s in enumerate(q):
        if s: n += 1
        else:
            if n >= 16 and k - n > 5: out.append((i0 + k - n / 2) / 100)
            n = 0
    return out

ABBR = ("St.", "O.K.", "U.S.", "O.", "Mt.", "Dr.", "Jr.", "Sr.", "U S")
def sentences(t):
    if LANG in CJK and re.search(r"[。？！]", t): return [x for x in re.findall(r"[^。？！]+[。？！」』]*", t.strip()) if x.strip()]
    parts = [x for x in re.split(r"(?<=[.?!»\"”])\s+(?=[A-ZÁÉÍÓÚÑ¿«\"“])", t.strip()) if x]; out = []
    for x in parts:
        if out and out[-1].endswith(ABBR): out[-1] += " " + x
        else: out.append(x)
    return out

def speech_bounds(a, b):
    i0, i1 = int(a * 100), int(b * 100); v = np.where(env[i0:i1] >= thr)[0]
    return (a, b) if not len(v) else ((i0 + v[0]) / 100, (i0 + v[-1] + 1) / 100)

def fits(t, width=42, lines=2): return len(wrap(t, width)) <= lines
def chunks(t, width=42, lines=2):
    """balanced cues of <= 2 wrapped lines (no tiny tails)"""
    if fits(t, width, lines): return [t]
    if LANG in CJK:   # split at commas / mid-points into balanced pieces of <= 2 lines
        n = 2
        while True:
            target = len(t) / n; out, cur = [], ""
            for ch in t:
                cur += ch
                if len(out) < n - 1 and len(cur) >= target and (ch in "，、；：" or len(cur) >= target * 1.15): out.append(cur); cur = ""
            if cur: out.append(cur)
            if all(fits(c) for c in out): return out
            n += 1
    words = t.split()
    n = 2
    while True:
        target = len(t) / n; out, cur = [], ""
        for w in words:
            cand = (cur + " " + w).strip()
            if cur and len(out) < n - 1 and (len(cand) > target * 1.08 or not fits(cand, width, lines)):
                out.append(cur); cur = w
            else: cur = cand
        out.append(cur)
        if all(fits(c, width, lines) for c in out): return out
        n += 1

def anchors(a, b, en_sents):
    """sentence start/end times anchored to pauses; fall back to char share"""
    s0, s1 = speech_bounds(a, b); P = pauses(s0, s1); n = len(en_sents)
    L = np.cumsum([len(x) for x in en_sents]); tot = L[-1]; cuts = []
    for k in range(n - 1):
        guess = s0 + (s1 - s0) * L[k] / tot
        cand = [p for p in P if p not in cuts and (not cuts or p > cuts[-1])]
        best = min(cand, key=lambda p: abs(p - guess)) if cand else None
        cuts.append(best if best is not None and abs(best - guess) < 2.5 else guess)
    return [s0] + cuts + [s1]

def build(lang):
    cues = []
    for it in tl:

            txt = (TR[lang] if lang != "en" else BITES_EN).get(it["id"]); a, b = speech_bounds(it["start"] + 0.2, it["start"] + it["dur"] - 0.2)
            if txt:
                cs = chunks(txt); n = len(cs)
                cues += [(a + (b - a) * i / n, a + (b - a) * (i + 1) / n, c) for i, c in enumerate(cs)]
            continue
        sid = it.get("id")
        if it["kind"] != "seg" or not disp.get(sid): continue
        en = disp[sid]
        for x, y in NUM: en = en.replace(x, y)
        en_s = sentences(en); spoken = sentences(it["text"])
        T = anchors(it["start"], it["start"] + it["dur"], spoken if len(spoken) == len(en_s) else en_s)
        sents = en_s if lang == "en" else sentences(TR[lang][sid])
        if len(sents) != len(en_s):   # translation regrouped sentences: spread by char share over the whole line
            T = [T[0], T[-1]]; sents = [("" if lang in CJK else " ").join(sents)]
        for k, s in enumerate(sents):
            a, b = T[k], T[k + 1]; cs = chunks(s); tot = sum(len(c) for c in cs); t = a
            for c in cs:
                d = (b - a) * len(c) / tot; cues.append((t, t + d, c)); t += d
    # merge cues that would flash (< 1.3 s) or are a lone word into a neighbour when the text still fits two lines
    merged = []; sep = "" if lang in CJK else " "
    for a, b, c in cues:

            pa, pb, pc = merged[-1]; merged[-1] = (pa, b, pc + sep + c); continue

            pa, pb, pc = merged[-1]; merged[-1] = (pa, b, pc + sep + c); continue
        merged.append((a, b, c))

    # hygiene: no overlaps, min 1.0 s on screen where room allows, small gap
    fixed = []

        b = min(max(b, a + 1.0), nxt - 0.04)
        fixed.append((a, b, c))
    return fixed

import numpy as np, soundfile as sf, json
SR=48000; W=SR//2
def rms_db(p):
    a,_=sf.read(p,dtype='float32',always_2d=True); a=a.mean(1); n=len(a)//W
    r=np.sqrt((a[:n*W].reshape(n,W)**2).mean(1))+1e-7; return 20*np.log10(r)
v=rms_db('stem_voice.wav'); m=rms_db('stem_music.wav'); am=rms_db('stem_amb.wav'); sx=rms_db('stem_sfx.wav')
n=min(map(len,(v,m,am,sx))); v,m,am,sx=v[:n],m[:n],am[:n],sx[:n]
tl=json.load(open('timeline.json')); ins=np.zeros(n,bool)
for it in tl:
    if it['kind']=='insert': ins[int(it['start']*2):int((it['start']+it['dur'])*2)]=True
spk=(v>-35)&~ins; gap=(v<-50)&~ins
print("voice-music under speech median %.1f dB (p10 %.1f)"%(np.median(v[spk]-m[spk]),np.percentile(v[spk]-m[spk],10)))
print("voice-amb under speech median %.1f dB"%np.median(v[spk]-am[spk]))
print("narr voice %.1f, insert voice %.1f"%(np.median(v[spk]),np.median(v[ins&(v>-40)])))
print("music in gaps %.1f"%np.median(m[gap]))
