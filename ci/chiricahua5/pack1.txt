#@@FILE chiricahua/render_v2.py
"""Render episode v2: shots_v2.json -> video; narration(+inserts) + music bed + ambience + SFX -> mix -> final mp4.
Shot extras: amb (wav path), amb_g (dB), sfx [{"f": wav, "at": sec from shot start, "g": dB}], fin/fout fades.
Audio rules: voice always on top; music & ambience duck under voice; inserts (interviews/news) get no SFX, music ~off."""
import json, os, sys, numpy as np, soundfile as sf
sys.path.insert(0, "/home/claude/doc/chiricahua"); import engine as E
D = "/home/claude/doc/chiricahua"; FPS = 30; SR = 48000
shots = json.load(open(os.environ.get("SHOTS", f"{D}/shots_v2.json"))); tl = json.load(open(f"{D}/timeline.json"))
TAG = os.environ.get("MIX_TAG", "")   # "" = English; "_es" = Spanish dub (audio only)
narr, _ = sf.read(f"{D}/narration_full{TAG}.wav", dtype="float32"); total = len(narr) / SR
vmask = np.load(f"{D}/voice_mask{TAG}.npy").astype(np.float32)  # 100 Hz, 1 where any voice

# ---------- frame-accurate timing ----------
edges = [0.0]
for s in shots: edges.append(edges[-1] + s["dur"])
edges[-1] = total
fr = [round(e * FPS) for e in edges]
for i, s in enumerate(shots):
    s["dur"] = (fr[i + 1] - fr[i]) / FPS; s["_start"] = fr[i] / FPS

# ---------- video ----------
work = f"{D}/render_v2"; os.makedirs(work, exist_ok=True)
only = sys.argv[1] if len(sys.argv) > 1 else None
sel = range(*map(int, only.split(":"))) if only and ":" in only else range(len(shots))
os.environ.setdefault("ENGINE_CACHE", f"{D}/cache")
files = []
for i in (sel if only != "audio" else []):
    s = {k: v for k, v in shots[i].items() if not k.startswith("_") and k not in ("amb", "amb_g", "sfx", "tag")}
    f = f"{work}/{i:04d}_{E.key(s)}.mp4"
    if s.get("src") and not os.path.exists(s["src"]):
        print("SKIP_MISSING", i, s["src"], flush=True); files.append(f); continue
    if not os.path.exists(f):                       # same shot rendered before at another index: reuse it (hard link)
        import glob as _g
        prev = [p for p in _g.glob(f"{work}/*_{E.key(s)}.mp4") if os.path.getsize(p) > 0]
        if prev:
            try: os.link(prev[0], f)
            except OSError: import shutil as _sh; _sh.copy(prev[0], f)
    if not os.path.exists(f):
        try: E.render_shot(s, f)
        except Exception as e:
            print("FAIL", i, s.get("src"), str(e)[-300:], flush=True)
            E.render_shot({"type": "card", "style": "date", "title": "", "dur": s["dur"]}, f)
    files.append(f)
    if i % 10 == 0: print("shot", i, "/", len(shots), flush=True)
if os.environ.get("VIDEO_ONLY"): print("VIDEO_RANGE_DONE", flush=True); sys.exit(0)   # parallel partial video passes
if only != "audio": open(f"{work}/concat.txt", "w").write("".join(f"file '{f}'\n" for f in files))
if only == "video" or (only != "audio" and any(not os.path.exists(f) for f in files)): print("VIDEO_PASS_DONE (missing shots remain)" if any(not os.path.exists(f) for f in files) else "VIDEO_PASS_DONE"); sys.exit(0)

# ---------- audio ----------
N = int(total * SR); hop = SR // 100
def env100(mask100):  # 100 Hz -> sample rate, smoothed
    m = np.convolve(mask100, np.ones(30) / 30, mode="same")
    g = np.repeat(m.astype(np.float32), hop); return np.pad(g, (0, max(0, N - len(g))), mode="edge")[:N]
voice = env100(vmask[: N // hop + 1])
ins = np.zeros(N // hop + 1, np.float32)
for it in tl:
    if it["kind"] == "insert": ins[int(it["start"] * 100): int((it["start"] + it["dur"]) * 100)] = 1
ins = env100(ins)
cache = {}
def load(p):
    if p not in cache:
        a, sr = sf.read(p, dtype="float32", always_2d=True)
        if a.shape[1] == 1: a = np.repeat(a, 2, 1)
        if sr != SR:
            import subprocess
            raw = subprocess.run(["ffmpeg", "-v", "error", "-i", p, "-ac", "2", "-ar", str(SR), "-f", "f32le", "-"], capture_output=True).stdout
            a = np.frombuffer(raw, np.float32).reshape(-1, 2)
        cache[p] = a
    return cache[p]
def place(buf, a, t0, gain, fade=0.05):
    s = int(t0 * SR);
    if s >= N: return
    a = a[: N - max(s, 0)].copy() if s >= 0 else a[-s:N - 0].copy()
    s = max(s, 0); n = len(a)
    if n == 0: return
    k = min(int(fade * SR), n // 2)
    if k: r = np.linspace(0, 1, k, dtype=np.float32); a[:k] *= r[:, None]; a[-k:] *= r[::-1, None]
    buf[s:s + n] += a * gain

# music bed by chapter mood
MOOD = {"cold_open": "tension", "chapter_1": "wonder", "chapter_2": "tension", "chapter_3": "somber",
        "chapter_4": "western", "chapter_5": "somber", "chapter_6": "wonder", "close": "wonder"}
music = np.zeros((N, 2), np.float32)
spans = []; cur = "cold_open"; t0 = 0.0
for it in tl:
    ch = it.get("chapter", cur)
    if ch != cur: spans.append((cur, t0, it["start"])); cur = ch; t0 = it["start"]
spans.append((cur, t0, total))
import score as SC
for n, (ch, a, b) in enumerate(spans):
    # one original bed per chapter, as long as the chapter: no audible loop, no identical music restarting in a later chapter
    mp = f"{D}/music/span_{n:02d}_{ch}.wav"
    if not os.path.exists(mp):
        SC.rng = np.random.default_rng(101 + 17 * n); sf.write(mp, SC.build(MOOD.get(ch, "wonder"), b - a + 6.0), SR)
    loop = load(mp)[: int((b - a + 2.5) * SR)]
    place(music, loop, a, 1.0, fade=2.5)
g_music = (0.40 - 0.34 * voice) * (1 - 0.92 * ins)          # 0.40 in gaps, 0.06 under voice (~18-19 dB below narrator), ~0 under inserts
g_music *= 0.6; music *= g_music[:, None]; del g_music
mix = np.zeros((N, 2), np.float32)
def bank(name, layer):
    sf.write(f"{D}/stem_{name}{TAG}.wav", layer, SR); np.add(mix, layer, out=mix)
bank("music", music); del music

# ambience per shot (cross-faded), RMS-normalized
amb = np.zeros((N, 2), np.float32)
from scipy.signal import butter, sosfiltfilt
BED = {}
def bed(p):
    """clean ambience bed: high-pass 100 Hz (no rumble / hum), tame sudden events, and stretch short recordings to 120 s
    by shuffling crossfaded segments so nothing repeats on a beat; RMS-normalized."""
    if p in BED: return BED[p]
    a = sosfiltfilt(butter(4, 100, "high", fs=SR, output="sos"), load(p), axis=0).astype(np.float32)
    w = int(0.2 * SR); e = np.sqrt(np.convolve((a ** 2).mean(1), np.ones(w) / w, "same")) + 1e-9
    cap = 2.0 * np.median(e); a *= np.minimum(1.0, cap / e)[:, None]
    if len(a) < 60 * SR:
        r = np.random.default_rng(len(a)); seg = min(int(3.0 * SR), len(a) // 2); xf = int(0.8 * SR); out = [a[:seg]]
        while sum(len(x) for x in out) < 120 * SR:
            st = int(r.integers(0, len(a) - seg)); nx = a[st:st + seg].copy(); r_ = np.linspace(0, 1, xf, dtype=np.float32)[:, None]
            prev = out[-1]; prev[-xf:] = prev[-xf:] * (1 - r_) + nx[:xf] * r_; out.append(nx[xf:])
        a = np.concatenate(out)
    a = a / (float(np.sqrt(np.mean(a ** 2))) + 1e-9) * 0.035
    BED[p] = a; return a
for s in shots:
    if not s.get("amb") or not os.path.exists(s["amb"]): continue
    a = bed(s["amb"]); L = int((s["dur"] + 0.8) * SR); st = int((s["_start"] - 0.4) * SR)
    loop = a[(max(st, 0) + np.arange(L)) % len(a)]          # continuous bed: offset by timeline position, never restarts per shot
    place(amb, loop, s["_start"] - 0.4, 10 ** (s.get("amb_g", 0) / 20), fade=0.6)
amb *= ((1.0 - 0.72 * voice) * (1 - ins))[:, None]
bank("amb", amb); del amb

# SFX: per shot + automatic transition sounds
sfx = np.zeros((N, 2), np.float32)
X = f"{D}/sfx"
for s in shots:
    for e in s.get("sfx", []):
        if os.path.exists(e["f"]):
            a = load(e["f"]); pk = float(np.max(np.abs(a)) + 1e-9)
            place(sfx, a / pk * 0.5, s["_start"] + e.get("at", 0), 10 ** (e.get("g", -6) / 20), fade=0.03)
    if s.get("type") == "card":
        place(sfx, load(f"{X}/whoosh.wav") * 0.55, s["_start"] - 0.55, 1.0)
        place(sfx, load(f"{X}/hit_low.wav") * 0.7, s["_start"] + 0.15, 1.0)
    if s.get("date"):
        place(sfx, load(f"{X}/type.wav"), s["_start"] + 0.25, 0.5)
sfx *= (1 - ins)[:, None]
bank("sfx", sfx); del sfx; cache.clear()

nm = narr if narr.ndim == 1 else narr.mean(1)
nm = np.pad(nm[:N], (0, max(0, N - len(nm))))
sf.write(f"{D}/stem_voice{TAG}.wav", nm.astype(np.float32), SR)
mix += nm.astype(np.float32)[:, None]; del nm
pk = float(np.max(np.abs(mix))); mix /= max(1.0, pk / 0.97)
sf.write(f"{D}/mix_v2{TAG}.wav", mix, SR, subtype="FLOAT")
import subprocess, re as _re
_m = subprocess.run(["ffmpeg", "-nostats", "-i", f"{D}/mix_v2{TAG}.wav", "-af", "loudnorm=I=-14:TP=-1.5:LRA=11:print_format=json", "-f", "null", "-"], capture_output=True, text=True).stderr
_j = json.loads(_m[_m.rindex("{"):_m.rindex("}") + 1])
LN2 = f"loudnorm=I=-14:TP=-1.5:LRA=11:measured_I={_j['input_i']}:measured_TP={_j['input_tp']}:measured_LRA={_j['input_lra']}:measured_thresh={_j['input_thresh']}:offset={_j['target_offset']}:linear=true"
E.run(["ffmpeg", "-v", "error", "-y", "-i", f"{D}/mix_v2{TAG}.wav", "-af", LN2, "-ar", "48000", f"{D}/mix_v2{TAG}_norm.wav"])
out = os.environ.get("FINAL_OUT", f"{D}/chiricahua_ep1_v2.mp4")
if not only:
    E.final(f"{work}/concat.txt", f"{D}/mix_v2{TAG}_norm.wav", out); print("FINAL", out)
else:
    print("partial ok")
