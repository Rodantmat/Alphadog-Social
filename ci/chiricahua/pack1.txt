_len = {}
def plen(p):
    if p not in _len:
        o = subprocess.run(["ffprobe", "-v", "error", "-show_entries", "format=duration", "-of", "csv=p=0", p], capture_output=True, text=True).stdout
        _len[p] = float(o or 0)
    return _len[p]
def tsec(t): m, s = t.split(":"); return int(m) * 60 + int(s)

def SFX(name, at=0.0, g=-8): return {"f": f"{X}/{name}.wav", "at": at, "g": g}
GRADE_FLY = "eq=brightness=-0.08:contrast=1.25:saturation=0.95:gamma=0.85,colorbalance=rm=-0.08:gm=0.0:bm=0.04:rh=-0.03:bh=0.04"
PERIOD = "hue=s=0,eq=contrast=1.12:brightness=-0.03,colorchannelmixer=.95:.05:0:0:.05:.9:.05:0:0:.1:.8,noise=alls=10:allf=t"   # modern steam-train stock graded to sit with 1880s archival

def Y(vid, sheet_t, off=1.5, **kw):
    """YouTube CC cut: sheet_t = timestamp seen on the review sheet; piece starts 6 s before it."""
    ss = max(0, tsec(sheet_t) - 6) if isinstance(sheet_t, str) else sheet_t
    if not os.path.exists(f"{YT}/yt_{vid}_{ss}.mp4"):   # map the timestamp into whichever downloaded piece contains it (else the nearest)
        import glob as _g
        T = ss + off; best = None
        for f in _g.glob(f"{YT}/yt_{vid}_*.mp4"):
            p0 = int(f.rsplit("_", 1)[1][:-4]); L = plen(f)
            d = 0 if p0 <= T < p0 + L else min(abs(T - p0), abs(T - (p0 + L)))
            if best is None or d < best[0]: best = (d, p0, L)
        if best: ss, off = best[1], min(max(0.0, T - best[1]), max(0.0, best[2] - 1))
    s = {"type": "video", "src": f"{YT}/yt_{vid}_{ss}.mp4", "ss": off, "drift": 0.03}
    if vid == "LNceEWbCiAg": s["vf_extra"] = "hue=s=0.45,colorbalance=rm=0.05:bm=-0.06:rh=0.04:bh=-0.08,eq=contrast=1.08:brightness=-0.04"  # muted, warm: kills the dashcam's cyan sky
    s.update(kw); return s
def S(name, off=0.5, **kw):
    s = {"type": "video", "src": f"{ST}/{name}.mp4", "ss": off, "drift": 0.03}; s.update(kw); return s
def F(n, ss=0, **kw):
    s = {"type": "video", "src": f"{FLY}/{n}.mp4", "ss": ss, "drift": 0.0, "vf_extra": GRADE_FLY}; s.update(kw); return s
def MAP(n, **kw): s = {"type": "map", "src": f"{D}/maps/{n}.mp4"}; s.update(kw); return s
def P(i, **kw):
    s = {"type": "photo", "src": f"{M}/{IM[i]['file']}", "z0": 1.0, "z1": 1.07, "x0": .5, "y0": .5, "x1": .5, "y1": .47, "bg": "fit"}; s.update(kw); return s
def R(i, fit=False, **kw):
    s = {"type": "photo", "src": f"{D}/ranch/{RL[i]['file']}", "z0": 1.0, "z1": 1.08, "x0": .5, "y0": .5, "x1": .52, "y1": .48}
    if fit: s["bg"] = "fit"
    s.update(kw); return s
def AMB(shots, key, g=0):
    for s in shots: s.setdefault("amb", A[key]); s.setdefault("amb_g", g)
    return shots

def calm(s):
    b = os.path.basename(s.get("src", ""))
    return next((v for k, v in CALM.items() if b.startswith(k)), None)

def spans(s):
    """cut-free usable spans (source seconds) inside the reviewed SAFE windows (rules V4, V10)"""
    src = s["src"]; L = plen(src); key = os.path.basename(src)[:-4]
    wins = SAFE.get(key, [(0, L)])
    cuts = sorted(CUTS.get(src, [])); out = []
    for a, b in wins:
        b = min(b, L - 0.15); pts = [a] + [c for c in cuts if a + 0.05 < c < b - 0.05] + [b]
        for x, y in zip(pts, pts[1:]):
            x2 = x + (0.1 if x != a else 0); y2 = y - (0.1 if y != b else 0)
            if y2 - x2 >= 1.0: out.append((x2, y2))
    return out

def capacity(s):
    if s["type"] != "video": return 1e9
    sp = spans(s); minsp = 0.5 if calm(s) else 0.8
    return max((b - a) / minsp for a, b in sp) if sp else 0.0

def place_video(s, dur, used=None):
    """pick a cut-free span (prefer the one holding the planned in-point), set ss / speed; returns the duration it can fill"""
    sp = spans(s)
    if not sp: SAFE_LOG.append(("NO_SPAN", os.path.basename(s["src"]))); return 0.0
    want = calm(s) or 1.0; minsp = 0.5 if calm(s) else 0.8
    home = next((w for w in sp if w[0] <= s.get("ss", 0) < w[1]), None)
    used = used if used is not None else set()
    cands = ([home] if home else []) + sorted(sp, key=lambda w: -(w[1] - w[0]))
    fresh = [w for w in cands if (s["src"], w) not in used]
    cands = fresh or cands
    for a, b in cands:
        room = b - a
        if dur * want <= room:
            s["ss"] = round(min(max(s.get("ss", a), a), b - dur * want), 3); sp_ = want; break
        if dur * minsp <= room:
            s["ss"] = round(a, 3); sp_ = room / dur; break
    else:
        a, b = cands[0] if home else max(sp, key=lambda w: w[1] - w[0]); s["ss"] = round(a, 3); sp_ = minsp; dur = (b - a) / minsp
    used.add((s["src"], next(w for w in sp if w[0] <= s["ss"] + 0.01 < w[1] + 0.01)))
    if sp_ < 0.999: s["speed"] = round(sp_, 3)
    else: s.pop("speed", None)
    if calm(s) and INTERP not in s.get("vf_extra", ""):
        s["vf_extra"] = (s["vf_extra"] + "," if s.get("vf_extra") else "") + INTERP
    key = os.path.basename(s["src"])[:-4]
    if key == "yt_SGEI69bBs6o_114" and not (s["ss"] < 8 - dur or 17 <= s["ss"] <= 25 - dur):
        s["vf_extra"] = "crop=iw*0.5:ih*0.5:iw*0.25:ih*0.25"
    return dur

def allocate(vis, dur, minshot=None):
    """rules V5/V6: maps play their whole animation + hold; everything else shares the rest, 3.5-9 s each, never past a source cut"""
    maps = [v for v in vis if v["type"] == "map"]
    for m in maps: m["dur"] = min(plen(m["src"]) + MAPHOLD, dur * 0.8 if len(vis) > 1 else dur)
    rest = [v for v in vis if v["type"] != "map"]; left = dur - sum(m["dur"] for m in maps)
    while len(rest) > 1 and left / len(rest) < (minshot or MINSHOT): rest.pop()          # too many visuals for the time: keep the first ones
    vis[:] = [v for v in vis if v in maps or v in rest]
    if not rest:
        k = left / max(1, len(maps))
        for m in maps: m["dur"] += k
        return
    for _ in range(6):                                                          # share time, respecting each clip's cut-free capacity
        free = [v for v in rest if "_cap" not in v]; fixed = sum(v["dur"] for v in rest if "_cap" in v)
        if not free: break
        W = sum(v.get("w", 1) for v in free); unit = (left - fixed) / W; capped = False
        for v in free:
            c = capacity(v); want = unit * v.get("w", 1)
            if c < want: v["dur"] = c; v["_cap"] = 1; capped = True
            else: v["dur"] = want
        if not capped: break
    short = left - sum(v["dur"] for v in rest)
    if short > 0.05:                                                            # every clip at capacity: give the rest to stills/maps
        hold = [v for v in vis if v["type"] in ("photo", "map")] or [max(rest, key=lambda v: v["dur"])]
        for v in hold: v["dur"] += short / len(hold)
    used = set()
    for v in rest:
        v.pop("_cap", None)
        if v["type"] == "video": v["dur"] = place_video(v, v["dur"], used)
    for _ in range(4):                                    # hand any leftover only to shots that have room (stills, maps, longer spans)
        left2 = dur - sum(v["dur"] for v in vis)
        if abs(left2) < 0.01: break
        room = {id(v): (1e9 if v["type"] != "video" else capacity(v) - v["dur"]) for v in vis}
        takers = [v for v in vis if room[id(v)] > 0.05] if left2 > 0 else [v for v in vis if v["dur"] > MINSHOT]
        if not takers: SAFE_LOG.append(("NO_ROOM", [os.path.basename(v.get("src", "")) for v in vis], round(left2, 2))); takers = [max(vis, key=lambda v: v["dur"])]
        for v in takers:
            add = left2 / len(takers)
            if v["type"] == "video" and left2 > 0: add = min(add, room[id(v)])
            v["dur"] += add
            if v["type"] == "video": used.discard(next(((v["src"], w) for w in spans(v) if w[0] <= v["ss"] + 0.01 < w[1] + 0.01), None)); v["dur"] = place_video(v, v["dur"], used)
    for v in vis: v["dur"] = round(v["dur"], 3); v.pop("w", None)
    drift = dur - sum(v["dur"] for v in vis)
    tgt = max(vis, key=lambda v: (1e9 if v["type"] != "video" else capacity(v) - v["dur"])); tgt["dur"] = round(tgt["dur"] + drift, 3)
