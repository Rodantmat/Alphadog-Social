#@@FILE chiricahua/validate_shots.py
"""Delete rendered shot files whose duration doesn't match the plan (partial writes from interrupted renders)."""
import json, os, subprocess, sys
sys.path.insert(0, "/home/claude/doc/chiricahua"); import engine as E, soundfile as sf
D = "/home/claude/doc/chiricahua"; shots = json.load(open(f"{D}/shots_v2.json"))
total = sf.info(f"{D}/narration_full.wav").duration; edges = [0.0]
for s in shots: edges.append(edges[-1] + s["dur"])
edges[-1] = total; fr = [round(e * 30) for e in edges]; bad = 0; ok = 0; miss = 0
for i, s in enumerate(shots):
    s["dur"] = (fr[i + 1] - fr[i]) / 30
    f = f"{D}/render_v2/{i:04d}_{E.key({k: v for k, v in s.items() if not k.startswith('_') and k not in ('amb', 'amb_g', 'sfx', 'tag')})}.mp4"
    if not os.path.exists(f): miss += 1; continue
    d = subprocess.run(["ffprobe", "-v", "error", "-show_entries", "format=duration", "-of", "csv=p=0", f], capture_output=True, text=True).stdout.strip()
    try: d = float(d)
    except ValueError: d = -1
    if abs(d - s["dur"]) > 0.12: print("BAD", i, f, d, s["dur"]); os.remove(f); bad += 1
    else: ok += 1
print("ok", ok, "bad", bad, "missing", miss)

#@@FILE fit_narr.py
"""Rebuild time-fitted narration WAVs: a re-voiced line fitted to the exact length of the line it replaced."""
import json, subprocess, numpy as np, soundfile as sf
SR = 48000; D = "/home/claude/doc/chiricahua"
def dec(p, af=None):
    return np.frombuffer(subprocess.run(["ffmpeg", "-v", "error", "-i", p] + (["-af", af] if af else []) + ["-f", "f32le", "-ac", "1", "-ar", str(SR), "-"], capture_output=True).stdout, np.float32)
for it in json.load(open(f"{D}/narr_fit.json")):
    old = dec(it["ref"]); new = dec(it["src"])
    if it["method"] == "pad" and len(new) <= len(old): out = np.pad(new, (0, len(old) - len(new)))
    else:
        out = dec(it["src"], f"atempo={len(new) / len(old):.6f}"); out = np.pad(out, (0, max(0, len(old) - len(out))))[:len(old)]
    sf.write(f"{D}/narr/{it['id']}.wav", out, SR, subtype="FLOAT"); print("fit", it["id"], len(old))

#@@FILE gate.py
"""The runner build must reproduce the approved cut exactly: compare timeline + shot plan against fingerprint.json."""
import json, hashlib, os
os.chdir("/home/claude/doc/chiricahua")
def h(x): return hashlib.sha1(json.dumps(x, sort_keys=True).encode()).hexdigest()[:8]
fp = json.load(open("fingerprint.json")); tl = json.load(open("timeline.json")); sh = json.load(open("shots_v2.json"))
t = [h([it.get("id"), round(it["start"], 3), round(it["dur"], 3)]) for it in tl]
s = [h([x.get("src"), round(x["dur"], 3), x.get("t0"), x.get("type")]) for x in sh]
bt = [i for i, (x, y) in enumerate(zip(t, fp["timeline"])) if x != y]; bs = [i for i, (x, y) in enumerate(zip(s, fp["shots"])) if x != y]
print("timeline", len(t), "diffs", bt[:10], [(tl[i].get("id"), tl[i]["start"], tl[i]["dur"]) for i in bt[:5]])
print("shots", len(s), "diffs", bs[:20], [(sh[i].get("src"), sh[i]["dur"]) for i in bs[:5]])
assert len(t) == len(fp["timeline"]) and len(s) == len(fp["shots"]) and not bt and not bs, "build differs from the approved cut"
print("GATE_OK")
